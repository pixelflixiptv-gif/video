const ax=(n,d,dur,o)=>{ o=o||{}; return `animation:${n} ${dur}s ${o.ease||'ease-out'} ${d}s ${o.it||1} ${o.dir||'normal'} both;`; };
const SH=(inner,on,off,anim,cls)=>`<g style="${vis(on||0,off)}">${anim?`<g class="${cls||'gx'}" style="${anim}">${inner}</g>`:inner}</g>`;
const TXT=(s,x,y,sz,fill,o)=>{ o=o||{}; const on=o.on||0;
  const t=`<text x="${x}" y="${y}" text-anchor="${o.anchor||'middle'}" font-family="Impact,'Arial Black',sans-serif" font-size="${sz}" fill="${fill}" stroke="${o.stroke||INK}" stroke-width="${o.sw!=null?o.sw:Math.round(sz*.12)}" stroke-linejoin="round" paint-order="stroke" style="letter-spacing:2px"${o.rot?` transform="rotate(${o.rot} ${x} ${y})"`:''}>${s}</text>`;

const HW=(s,x,y,sz,fill,on,o)=>{ o=o||{};
  return SH(`<text x="${x}" y="${y}" font-family="'Courier New',monospace" font-weight="700" font-size="${sz}" fill="${fill}">${s}</text>`,on,o.off,ax('gxIn',on,.3)); };

const BURST=(x,y,r,fill,o)=>{ o=o||{}; const n=o.n||14, on=o.on||0; let p='';
